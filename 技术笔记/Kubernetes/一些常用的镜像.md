## 如何部署一个 FTP 服务器
```bash
mkdir -p /data/ftp/data  && mkdir -p /data/ftp/log  && chmod -R 777 /data/ftp
# 启动FTP服务
docker run -d \
--name vsftpd   \
--restart=always \
-v /data/ftp/data/:/home/vsftpd \
-v  /data/ftp/log/:/var/log/vsftpd/ \
-p 20:20 -p 21:21 -p 20000:20000 \
-e FTP_USER=admin \
-e FTP_PASS=admin \
-e LOG_STDOUT=1 \
fauria/vsftpd
```
