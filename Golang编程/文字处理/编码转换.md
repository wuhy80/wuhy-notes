import "golang.org/x/text/encoding/simplifiedchinese"

func convertToUTF8(s string) string {
    decoder := simplifiedchinese.GB18030.NewDecoder()
    utf8Str, _ := decoder.String(s)
    return utf8Str
}

// 下面是使用convert接口
output, err := cmd.CombinedOutput()
if err != nil {
    comm.Log.Infof("ping cmd:%s res %+v", cmd, err)
    c.JSON(http.StatusInternalServerError, gin.H{"error": err.Error()})
    return
}

lines := strings.Split(string(output), "\n")

// 将内容转换为 UTF-8 编码
utf8Lines := make([]string, len(lines))
for i, line := range lines {
    utf8Lines[i] = convertToUTF8(line)
}

c.JSON(http.StatusOK, map[string]interface{}{"code": 0, "msg": "success", "data": utf8Lines})
