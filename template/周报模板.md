<%*
const fileTitle = tp.file.title
const folderParts = (tp.file.folder(true) || "").split("/")
const yearFromFolder = folderParts.find(part => /^\d{4}$/.test(part))

let weekStart = null

if (/^\d{4}-W\d{1,2}$/.test(fileTitle)) {
  weekStart = moment(fileTitle + "-1", "GGGG-[W]WW-E", true)
} else {
  const weekMatch = fileTitle.match(/^第(\d{1,2})周$/)
  if (weekMatch && yearFromFolder) {
    weekStart = moment().isoWeekYear(Number(yearFromFolder)).isoWeek(Number(weekMatch[1])).startOf("isoWeek")
  }
}

if (!weekStart || !weekStart.isValid()) {
  weekStart = moment().startOf("isoWeek")
}

weekStart = weekStart.startOf("isoWeek")

const weekEnd = weekStart.clone().endOf("isoWeek")
const weekDate = weekStart.format("YYYY-MM-DD")
const weekWeekday = weekStart.clone().locale("zh-cn").format("dddd")

const start = new Date("<% weekStart.format("YYYY-MM-DD") %>T00:00:00");
const end = new Date("<% weekEnd.format("YYYY-MM-DD") %>T23:59:59");

const weeklyDiaries = dv.pages()
  .where(page => page.file.path.startsWith("我的日记/"))
  .where(page => /^\d{4}-\d{2}-\d{2}$/.test(page.file.name))
  .where(page => {
    const date = new Date(`${page.file.name}T00:00:00`);
    return date >= start && date <= end;
  })
  .sort(page => page.file.name, "asc");
