# 本月日记

```dataviewjs
const start = new Date("<% monthStart.format("YYYY-MM-DD") %>T00:00:00");
const end = new Date("<% monthEnd.format("YYYY-MM-DD") %>T23:59:59");

const monthlyDiaries = dv.pages()
  .where(page => page.file.path.startsWith("我的日记/"))
  .where(page => /^\d{4}-\d{2}-\d{2}$/.test(page.file.name))
  .where(page => {
    const date = new Date(`${page.file.name}T00:00:00`);
    return date >= start && date <= end;
  })
  .sort(page => page.file.name, "asc");

dv.list(monthlyDiaries.map(page => page.file.link));
```
